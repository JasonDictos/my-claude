cmake_minimum_required(VERSION 3.20)
project(mc-term LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Svg Test DBus)
find_package(qtermwidget6 REQUIRED)
add_executable(mc-term src/main.cpp src/host.cpp src/form.cpp src/planpanel.cpp src/syskeys.cpp src/layericon.cpp src/tmuxctl.cpp src/dialect.cpp src/theme.cpp icons/icons.qrc)
target_link_libraries(mc-term PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg Qt6::Test Qt6::DBus qtermwidget6)
# Blur behind see-through terminals is asked of KWin with an X11 window property.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(XCB QUIET IMPORTED_TARGET xcb)
  if(XCB_FOUND)
    target_compile_definitions(mc-term PRIVATE MC_HAVE_XCB)
    if(MC_STATIC)
      target_link_libraries(mc-term PRIVATE ${XCB_STATIC_LDFLAGS})
    else()
      target_link_libraries(mc-term PRIVATE PkgConfig::XCB)
    endif()
  endif()
endif()

# MC_STATIC: one self-contained binary (packaging/static: musl, Qt and qtermwidget static).
# qtermwidget's keyboard layouts and colour schemes ride along as resources, unpacked at
# start (main.cpp), since a static build has no /usr/share/qtermwidget6 to read them from.
option(MC_STATIC "link everything in" OFF)
if(MC_STATIC)
  target_compile_definitions(mc-term PRIVATE MC_STATIC)
  target_link_options(mc-term PRIVATE -static)
  # what freetype and fontconfig need under them, last: a static link resolves left to right
  target_link_libraries(mc-term PRIVATE -Wl,--start-group fontconfig freetype brotlidec brotlicommon bz2 png z expat -Wl,--end-group)
  qt_import_plugins(mc-term INCLUDE Qt6::QXcbIntegrationPlugin Qt6::QOffscreenIntegrationPlugin Qt6::QSvgIconPlugin Qt6::QSvgPlugin)
  find_path(QTW_DATA kb-layouts PATHS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES share/qtermwidget6 NO_DEFAULT_PATH)
  file(GLOB_RECURSE QTW_FILES RELATIVE ${QTW_DATA} ${QTW_DATA}/kb-layouts/* ${QTW_DATA}/color-schemes/*)
  set(QRC "<RCC><qresource prefix=\"/qtw\">\n")
  foreach(f ${QTW_FILES})
    string(APPEND QRC "<file alias=\"${f}\">${QTW_DATA}/${f}</file>\n")
  endforeach()
  string(APPEND QRC "</qresource></RCC>\n")
  file(WRITE ${CMAKE_BINARY_DIR}/qtw.qrc "${QRC}")
  target_sources(mc-term PRIVATE ${CMAKE_BINARY_DIR}/qtw.qrc)
endif()
install(TARGETS mc-term RUNTIME DESTINATION bin)
install(FILES mc-term.desktop DESTINATION share/applications)
install(FILES icons/mc-term.svg DESTINATION share/icons/hicolor/scalable/apps)
foreach(n 16 22 24 32 48 64 128 256 512)
  install(FILES icons/mc-term-${n}.png DESTINATION share/icons/hicolor/${n}x${n}/apps RENAME mc-term.png)
endforeach()
